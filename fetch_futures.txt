"""Collect 1-minute CME index futures (NQ, ES, YM) from Yahoo Finance.

Yahoo only keeps 1-minute bars for the last ~30 days, so this runs daily
and appends to data/futures/<SYM>_1m.csv.gz (timestamps in UTC).
"""
import os
from datetime import datetime, timedelta, timezone

import pandas as pd
import yfinance as yf

SYMBOLS = {"NQ": "NQ=F", "ES": "ES=F", "YM": "YM=F"}
OUT = "data/futures"
os.makedirs(OUT, exist_ok=True)

now = datetime.now(timezone.utc)
start = now - timedelta(days=29)

for name, ticker in SYMBOLS.items():
    parts = []
    s = start
    while s < now:
        e = min(s + timedelta(days=7), now)
        try:
            df = yf.download(ticker, start=s, end=e, interval="1m",
                             progress=False, auto_adjust=False, prepost=True)
        except Exception as ex:  # keep going if one chunk fails
            print(f"{name} {s:%Y-%m-%d} failed: {ex}")
            df = pd.DataFrame()
        if not df.empty:
            if isinstance(df.columns, pd.MultiIndex):
                df.columns = df.columns.get_level_values(0)
            df = df[["Open", "High", "Low", "Close", "Volume"]]
            df.index = pd.to_datetime(df.index, utc=True)
            parts.append(df)
        s = e

    path = f"{OUT}/{name}_1m.csv.gz"
    if os.path.exists(path):
        old = pd.read_csv(path, index_col=0, parse_dates=True)
        old.index = pd.to_datetime(old.index, utc=True)
        parts.insert(0, old)
    if not parts:
        print(f"{name}: no data")
        continue
    all_ = pd.concat(parts)
    all_ = all_[~all_.index.duplicated(keep="last")].sort_index()
    all_.index.name = "Datetime_UTC"
    all_.to_csv(path, compression="gzip")
    print(f"{name}: {len(all_)} bars, {all_.index[0]} -> {all_.index[-1]}")
